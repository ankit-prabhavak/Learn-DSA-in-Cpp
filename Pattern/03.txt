ubuntu@ip-172-16-11-72:~$ #!/bin/bash

START_TIME=$(date -u -d "30 days ago" +%FT%TZ)
END_TIME=$(date -u +%FT%TZ)

printf "%-40s %-15s\n" "LOAD BALANCER NAME" "REQUEST COUNT"
printf "%s\n" "------------------------------------------------------------"

aws elbv2 describe-load-balancers \
  --query 'LoadBalancers[*].[LoadBalancerName,LoadBalancerArn]' \
  --output text | while read -r LB_NAME LB_ARN; do

    LB_RESOURCE_ID=$(echo "$LB_ARN" | sed 's/.*:loadbalancer\///')

    REQUEST_COUNT=$(aws cloudwatch get-metric-statistics \
        --namespace AWS/ApplicationELB \
        --metric-name RequestCount \
        --dimensions Name=LoadBalancer,Value="$LB_RESOURCE_ID" \
        --statistics Sum \
        --start-time "$START_TIME" \
        --end-time "$END_TIME" \
        --period 2592000 \
        --query 'sum(Datapoints[].Sum)' \
        --output text 2>/dev/null)

    REQUEST_COUNT=${REQUEST_COUNT:-0}

    if [[ "$REQUEST_COUNT" == "None" || "$REQUEST_COUNT" == "0" || "$REQUEST_COUNT" == "0.0" ]]; then
        printf "%-40s %-15s\n" "$LB_NAME" "0"
    fi

done
